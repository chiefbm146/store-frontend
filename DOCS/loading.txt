// NEW: Main initialization function
async function initializeCheckout() {
    // 1. Get Product ID from URL
    const urlParams = new URLSearchParams(window.location.search);
    const productId = urlParams.get('product_id') || '7-day-launchpad'; // Default to launchpad

    const productDetails = PRODUCT_DETAILS_MAP[productId];
    if (!productDetails) {
        showError("Invalid product selected. Please go back and choose a plan.");
        // NEW: Make sure the page is visible even on error
        document.querySelector('.checkout-wrapper').style.visibility = 'visible';
        return;
    }

    // 2. Update UI with product details (everything except price)
    document.querySelector('.product-name').textContent = productDetails.name;
    document.querySelector('.product-description').textContent = productDetails.description;
    const featuresList = document.querySelector('.features-list');
    featuresList.innerHTML = productDetails.features.map(feature => `
        <div class="feature-item">
            <span class="feature-icon">✓</span>
            <span class="feature-text"><strong>${feature}</strong></span>
        </div>
    `).join('');

    try {
        // 3. Create Payment Intent to get the correct price from the backend
        const affiliateRef = sessionStorage.getItem('affiliate_ref');
        const response = await fetch(`${BACKEND_URL}/api/v1/aarie-platform/create-payment-intent`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ product_id: productId, affiliate_ref: affiliateRef || null })
        });

        if (!response.ok) throw new Error('Could not initialize payment session.');
        
        const data = await response.json();
        clientSecret = data.clientSecret;
        const amount = data.amount;
        const displayPrice = `$${(amount / 100).toFixed(2).replace(/\.00$/, '')}`;

        // 4. Update UI with the dynamic price from the backend, removing the loading state
        const priceElement = document.querySelector('.product-price');
        priceElement.textContent = displayPrice;
        priceElement.style.animation = 'none'; // Stop the pulsing animation

        document.querySelector('.btn-text').textContent = 'Pay ' + displayPrice;
        
        // 5. Setup Stripe Elements now that we have the clientSecret
        cardElement.mount('#stripe-card-element');

    } catch (error) {
        showError(error.message);
        // In case of error, show a fallback price or message
        document.querySelector('.product-price').textContent = 'Error';
        document.querySelector('.btn-text').textContent = 'Unavailable';
        document.getElementById('submit-btn').disabled = true;
    }
}
Summary of the Fix:

HTML Change: We remove the hardcoded $99 and put a "..." placeholder that pulses visually. This immediately signals to the user that the final price is being calculated.

JavaScript Change: The script now fetches the correct price from the backend first. Once it receives the price, it updates both the price display and the payment button text, and then stops the loading animation.

This eliminates the price flicker entirely. The user will see a brief, professional loading animation for the price, which then resolves to the correct final amount. This is the standard and correct way to handle dynamic data on a checkout page.
