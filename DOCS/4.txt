Second, replace the entire JavaScript logic at the bottom of affiliate-aarie.html with this improved version. This new script correctly handles the different views.
code
Html
<!-- At the very bottom of affiliate-aarie.html -->
<script>
    document.addEventListener('DOMContentLoaded', function() {
        // Form submission logic
        const form = document.getElementById('affiliateForm');
        const submitBtn = document.getElementById('submitBtn');
        const errorDiv = document.getElementById('formError');
        const applicationFormView = document.getElementById('applicationFormView');
        const successView = document.getElementById('successView');
        const ageCheck = document.getElementById('ageCheck');
        const tosCheck = document.getElementById('tosCheck');

        function toggleSubmit() {
            submitBtn.disabled = !(ageCheck.checked && tosCheck.checked);
        }
        ageCheck.addEventListener('change', toggleSubmit);
        tosCheck.addEventListener('change', toggleSubmit);

        form.addEventListener('submit', async (e) => {
            e.preventDefault();
            submitBtn.disabled = true;
            submitBtn.textContent = 'Submitting...';
            errorDiv.style.display = 'none';

            const applicationData = {
                name: document.getElementById('name').value.trim(),
                email: document.getElementById('email').value.trim(),
                website: document.getElementById('website').value.trim(),
                message: document.getElementById('message').value.trim(),
                ageConfirmed: ageCheck.checked,
                tosAcceptedAt: new Date().toISOString()
            };

            try {
                const response = await fetch('/api/v1/aarie-platform/apply-for-affiliate', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(applicationData)
                });
                const data = await response.json();
                if (!response.ok) {
                    // Use the error message from the backend!
                    throw new Error(data.error || 'Something went wrong.');
                }

                applicationFormView.style.display = 'none';
                successView.style.display = 'block';
                successView.scrollIntoView({ behavior: 'smooth', block: 'center' });
                document.getElementById('goToPortalBtn').addEventListener('click', () => {
                     window.location.href = '/affiliate-portal.html';
                });

            } catch (error) {
                errorDiv.textContent = error.message;
                errorDiv.style.display = 'block';
                errorDiv.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
                submitBtn.disabled = false;
                submitBtn.textContent = 'Submit Application';
            }
        });

        // Toast notification logic
        const toast = document.getElementById('toastNotification');
        if (toast) {
            toast.addEventListener('click', () => { toast.style.display = 'none'; });
            setTimeout(() => { if (toast) toast.style.display = 'none'; }, 2500);
        }
    });

    // Separate logic for checking user status and showing the correct view
    document.addEventListener('DOMContentLoaded', () => {
        const loadingView = document.getElementById('loadingView');
        const applicationView = document.getElementById('applicationFormView');
        const alreadyAffiliateView = document.getElementById('alreadyAffiliateView');

        const onFirebaseReady = (callback) => {
            const interval = setInterval(() => {
                if (typeof firebase !== 'undefined' && firebase.auth) {
                    clearInterval(interval);
                    callback();
                }
            }, 100);
        };

        onFirebaseReady(() => {
            firebase.auth().onAuthStateChanged(async (user) => {
                if (user) {
                    // User is signed in, check status
                    try {
                        const idToken = await user.getIdToken(true);
                        const response = await fetch('/api/client/affiliate-status', {
                            headers: { 'Authorization': `Bearer ${idToken}` }
                        });
                        const data = await response.json();

                        if (data.isAffiliate) {
                            loadingView.style.display = 'none';
                            applicationView.style.display = 'none';
                            alreadyAffiliateView.style.display = 'block';
                        } else {
                            document.getElementById('name').value = user.displayName || '';
                            document.getElementById('email').value = user.email || '';
                            loadingView.style.display = 'none';
                            applicationView.style.display = 'block';
                        }
                    } catch (err) {
                        console.error("Status check failed:", err);
                        loadingView.style.display = 'none';
                        applicationView.style.display = 'block'; // Fallback to form
                    }
                } else {
                    // User not signed in, show the form
                    loadingView.style.display = 'none';
                    applicationView.style.display = 'block';
                }
            });
        });
    });
</script>
Why This New Solution Is Foolproof
Backend Authority: The server (main.py) now has the final say. Even if a user managed to bypass the frontend JavaScript, the backend would reject their application with a "409 Conflict" error.
Clear User Experience: The frontend (affiliate-aarie.html) now shows a "Checking status..." message first, then cleanly directs existing affiliates to their portal and new users to the application form, preventing any chance of confusion.
