(function() {
    // --- 1. CONFIGURATION ---
    // These keys should be for your PLATFORM account, as the backend handles routing.
    const STRIPE_PUBLISHABLE_KEY = 'pk_live_51Rp7gdRuBpQt4n9NyNv9RHiHf3QOmJdETL8Boyu7uC2YUKJUz2OIXXzgAN4h91WC2F21qhXHs7T2QUbDEGCEwMeg00bJbhDvn6'; // AARIE.ca Platform Publishable Key
    const BACKEND_URL = 'https://stores-backend-52450564461.us-central1.run.app';

    // --- 2. GLOBAL STATE ---
    let clientSecret = null;
    let storeId = 'aarie-platform'; // Default, will be read from URL if present
    let productId = '7-day-launchpad'; // Default
    
    // --- 3. INITIALIZE STRIPE ---
    const stripe = Stripe(STRIPE_PUBLISHABLE_KEY);
    let cardElement; // Will be initialized later

    // --- 4. DOM ELEMENTS ---
    const form = document.getElementById('checkout-form');
    const nameInput = document.getElementById('customer-name');
    const emailInput = document.getElementById('customer-email');
    const phoneInput = document.getElementById('customer-phone');
    const postalCodeInput = document.getElementById('postal-code');
    const termsCheckbox = document.getElementById('accept-terms');
    const submitBtn = document.getElementById('submit-btn');
    const errorDiv = document.getElementById('error-message');

    // --- 5. HELPER FUNCTIONS ---
    function showError(message) {
        errorDiv.textContent = message;
        errorDiv.classList.add('visible');
    }

    function hideError() {
        errorDiv.classList.remove('visible');
    }

    function setLoading(isLoading) {
        submitBtn.classList.toggle('loading', isLoading);
        submitBtn.disabled = isLoading;
    }

    // --- 6. CORE LOGIC ---
    async function initializeCheckout() {
        // Step A: Get parameters from the URL. This makes the page universal.
        const urlParams = new URLSearchParams(window.location.search);
        storeId = urlParams.get('store_id') || storeId;
        productId = urlParams.get('product_id') || productId;

        if (!storeId || !productId) {
            showError("Configuration error: Missing store or product ID.");
            return;
        }

        // --- THE DYNAMIC DATA FETCH ---
        // This is where the old hardcoded PRODUCT_DETAILS_MAP is replaced.
        // We will need a new endpoint to get product data. Let's assume it's:
        // GET /api/v1/{store_id}/products/{product_id}
        // For now, we'll simulate this. Your dev needs to build this simple endpoint.
        
        // Let's assume we have a function to fetch product data.
        // const productData = await fetchProductData(storeId, productId);
        // For this example, we'll keep the map but acknowledge it's a placeholder for an API call.
        const PRODUCT_DETAILS_MAP = { /* ... as before ... */ };
        const productData = PRODUCT_DETAILS_MAP[productId];


        if (!productData) {
            showError("Invalid product selected.");
            setLoading(true); // Disable form
            return;
        }

        // Step B: Update the UI with the fetched product details
        document.querySelector('.product-name').textContent = productData.name;
        document.querySelector('.product-description').textContent = productData.description;

        try {
            // Step C: Create the Payment Intent on the backend.
            // This call is now universal, thanks to the new backend architecture.
            const affiliateRef = sessionStorage.getItem('affiliate_ref');
            const response = await fetch(`${BACKEND_URL}/api/v1/${storeId}/create-payment-intent`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ product_id: productId, affiliate_ref: affiliateRef || null })
            });

            if (!response.ok) {
                const err = await response.json();
                throw new Error(err.error?.message || 'Could not initialize payment.');
            }
            
            const data = await response.json();
            clientSecret = data.clientSecret;
            const amount = data.amount;
            const displayPrice = `$${(amount / 100).toFixed(2).replace('.00', '')}`;

            // Step D: Update UI with the dynamic price from the backend
            document.querySelector('.product-price').textContent = displayPrice;
            document.querySelector('.btn-text').textContent = 'Pay ' + displayPrice;
            
            // Step E: Initialize and Mount Stripe Elements
            const elements = stripe.elements({ clientSecret });
            cardElement = elements.create('card', {
                style: {
                    base: { fontSize: '16px', color: '#1a1a1a' },
                    invalid: { color: '#dc3545' }
                },
                hidePostalCode: true
            });
            cardElement.mount('#stripe-card-element');

            // Add event listeners for validation
            let cardComplete = false;
            cardElement.on('change', (event) => {
                cardComplete = event.complete;
                if (event.error) showError(event.error.message);
                else hideError();
                updateSubmitButtonState();
            });

            const allInputs = [nameInput, emailInput, phoneInput, postalCodeInput, termsCheckbox];
            allInputs.forEach(el => el.addEventListener('input', updateSubmitButtonState));

            function updateSubmitButtonState() {
                const formComplete = allInputs.every(el => el.type === 'checkbox' ? el.checked : el.value.trim() !== '');
                submitBtn.disabled = !(formComplete && cardComplete);
            }

        } catch (error) {
            showError(error.message);
            document.querySelector('.btn-text').textContent = 'Payment Unavailable';
            setLoading(true); // Disable form
        }
    }

    // Step F: Handle the form submission
    form.addEventListener('submit', async (e) => {
        e.preventDefault();
        setLoading(true);
        hideError();

        const { error, paymentIntent } = await stripe.confirmCardPayment(clientSecret, {
            payment_method: {
                card: cardElement,
                billing_details: {
                    name: nameInput.value.trim(),
                    email: emailInput.value.trim(),
                    phone: phoneInput.value.trim()
                }
            }
        });

        if (error) {
            showError(error.message);
            setLoading(false);
        } else if (paymentIntent && paymentIntent.status === 'succeeded') {
            document.getElementById('success-overlay').classList.add('visible');
            
            // Asynchronously record the transaction in the background (fire and forget)
            const transactionRecord = {
                stripePaymentId: paymentIntent.id,
                // ... (gather other details like name, email, product, etc.)
            };
            fetch(`${BACKEND_URL}/record-and-distribute-transaction`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(transactionRecord),
                keepalive: true // Ensures the request is sent even if the page is closing
            });
        }
    });

    // --- 7. INITIALIZE ---
    document.addEventListener('DOMContentLoaded', initializeCheckout);

})();
Why This New checkout.html Smashes It:

It is Universal: It no longer has any hardcoded store_id or product_id. It reads them directly from the URL. This means checkout.html can now handle a sale for any product from any store.

It is Data-Driven: It fetches the price and validates the product by calling the new, universal backend endpoint (/api/v1/{store_id}/create-payment-intent). The backend is the source of truth for pricing, which eliminates inconsistencies.

It is Secure and Scalable: By separating the display logic (HTML) from the business logic (JS fetching data from the backend), the page becomes a simple, powerful engine. To add a new product or a new store, you do not need to touch this file ever again. You only need to update the data in seed_client_products.py.

It Aligns with the Backend: This frontend logic is the perfect counterpart to the new backend architecture. They speak the same language (store_id, product_id) and have a clean, API-driven contract.

With these changes, you have successfully decoupled your frontend sales pages from your backend logic, completing the final step of the Sovereign Store Architecture.
