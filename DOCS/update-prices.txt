// NEW: Main initialization function
async function initializeCheckout() {
    // 1. Get Product ID from URL
    const urlParams = new URLSearchParams(window.location.search);
    const productId = urlParams.get('product_id') || '7-day-launchpad'; // Default to launchpad

    const productDetails = PRODUCT_DETAILS_MAP[productId];
    if (!productDetails) {
        showError("Invalid product selected. Please go back and choose a plan.");
        return;
    }

    // 2. Update UI with product details
    document.querySelector('.product-name').textContent = productDetails.name;
    document.querySelector('.product-description').textContent = productDetails.description;
    const featuresList = document.querySelector('.features-list');
    featuresList.innerHTML = productDetails.features.map(feature => `
        <div class="feature-item">
            <span class="feature-icon">✓</span>
            <span class="feature-text"><strong>${feature}</strong></span>
        </div>
    `).join('');

    try {
        // 3. Create Payment Intent to get the price from backend
        const affiliateRef = sessionStorage.getItem('affiliate_ref');
        const response = await fetch(`${BACKEND_URL}/api/v1/aarie-platform/create-payment-intent`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ product_id: productId, affiliate_ref: affiliateRef || null })
        });

        if (!response.ok) throw new Error('Could not initialize payment.');
        
        const data = await response.json();
        clientSecret = data.clientSecret;
        const amount = data.amount;
        const displayPrice = `$${(amount / 100).toFixed(2).replace('.00', '')}`;

        // 4. Update UI with the dynamic price from backend
        document.querySelector('.product-price').textContent = displayPrice;
        document.querySelector('.btn-text').textContent = 'Pay ' + displayPrice;
        
        // 5. Setup Stripe Elements now that we have the clientSecret
        cardElement.mount('#stripe-card-element');

    } catch (error) {
        showError(error.message);
    }
}

// Attach event listeners for validation
cardElement.on('change', (event) => {
    cardComplete = event.complete;
    if (event.error) showError(event.error.message);
    else hideError();
    updateSubmitButton();
});

[nameInput, emailInput, phoneInput, postalCodeInput, termsCheckbox].forEach(el => {
    el.addEventListener('input', updateSubmitButton);
    el.addEventListener('change', updateSubmitButton);
});

// Handle form submission
document.getElementById('checkout-form').addEventListener('submit', async (e) => {
    e.preventDefault();
    setLoading(true);
    hideError();

    const { error } = await stripe.confirmCardPayment(clientSecret, {
        payment_method: {
            card: cardElement,
            billing_details: {
                name: nameInput.value.trim(),
                email: emailInput.value.trim(),
                phone: phoneInput.value.trim()
            }
        }
    });

    if (error) {
        showError(error.message);
        setLoading(false);
    } else {
        document.getElementById('success-overlay').classList.add('visible');
    }
});

// Run initialization on page load
document.addEventListener('DOMContentLoaded', initializeCheckout);
